# /sunny_ultimine config get/set (valid, unknown key, bad value, clamping), /sunny_ultimine status and shape, and
# /sunny_ultimine_dev status. max_blocks 32 must cap a real selection; every setting is put back at the end.
forceload add -16 -16 79 15
sunny_ultimine config get max_blocks
sunny_ultimine config set max_blocks 32
sunny_ultimine config get max_blocks
expect max_blocks = 64
expect max_blocks = 32
# Unknown keys and values that don't parse are refused and change nothing.
sunny_ultimine config get no_such_key
sunny_ultimine config set no_such_key 5
sunny_ultimine config set max_blocks lots
sunny_ultimine config set require_tool maybe
sunny_ultimine config get require_tool
expect Unknown setting "no_such_key"
expect Could not set no_such_key: unknown setting "no_such_key"
expect Could not set max_blocks: 
expect Could not set require_tool: 
expect require_tool = false
# Out-of-range numbers are clamped to the field's range.
sunny_ultimine config set prevent_tool_break 500
sunny_ultimine config set prevent_tool_break 0
expect prevent_tool_break = 100
expect prevent_tool_break = 0
# Tag lists take a bare comma-separated list.
sunny_ultimine config set merge_tags minecraft:logs, minecraft:*_ores
sunny_ultimine config get merge_tags
sunny_ultimine config set merge_tags minecraft:base_stone_overworld, minecraft:*_ores, c:ores/*, forge:ores/*
expect merge_tags = \["minecraft:logs", "minecraft:\*_ores"\]
expect merge_tags = \["minecraft:base_stone_overworld", "minecraft:\*_ores", "c:ores/\*", "forge:ores/\*"\]
