# All six shapes, broken by the dev fake player "tester" in survival with a diamond pickaxe. Every shape stands in
# the open air over a glass floor, so its exact block set is known; drops must arrive merged at the original block.
forceload add -16 -16 79 15
fill -8 96 -8 60 96 8 minecraft:glass
sunny_ultimine_dev gamemode tester survival
sunny_ultimine_dev give tester diamond_pickaxe
sunny_ultimine_dev food tester 20
sunny_ultimine_dev press tester true
expect DEV_GAMEMODE player=tester mode=survival ok=(true|false)
expect DEV_PRESS player=tester fake=(true|false) pressed=true

# 1. shapeless: diamond_ore + deepslate_diamond_ore (minecraft:diamond_ores), one joined diagonally; the stone next
#    to it is not in an ore tag and stays.
sunny_ultimine_dev shape tester shapeless
setblock 0 100 0 minecraft:diamond_ore
setblock 1 100 0 minecraft:diamond_ore
setblock 2 101 0 minecraft:deepslate_diamond_ore
setblock 3 101 0 minecraft:stone
sunny_ultimine_dev preview tester 0 100 0 up
sunny_ultimine_dev break tester 0 100 0 up
sunny_ultimine_dev drops 0 100 0 1
sunny_ultimine_dev drops 3 101 0 0
execute if block 0 100 0 minecraft:air if block 1 100 0 minecraft:air if block 2 101 0 minecraft:air run say S1 shapeless cluster gone
execute if block 3 101 0 minecraft:stone run say S1 stone kept
expect DEV_SHAPE player=tester shape=shapeless
expect DEV_PREVIEW player=tester pos=0 100 0 face=up can=Some\(Allowed\) matcher=ByTagsShapeless count=3 blocks=0,100,0;1,100,0;2,101,0$
expect DEV_BREAK ok=true player=tester pos=0 100 0 block=minecraft:diamond_ore can=Allowed
expect DEV_DROPS pos=0 100 0 item_entities=1 items=minecraft:diamondx3 xp_orbs=[1-9][0-9]* xp_total=([9]|[1-9][0-9])$
expect DEV_DROPS pos=3 101 0 item_entities=0 items=none xp_orbs=0 xp_total=0
expect \] S1 shapeless cluster gone
expect \] S1 stone kept
kill @e[type=item]
kill @e[type=experience_orb]
